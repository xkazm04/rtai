import { execFileSync } from 'node:child_process';
import { copyFileSync, mkdirSync, readdirSync, readFileSync, symlinkSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { runGodot } from '../../tools/lib/godot.mjs';
const out = process.argv[2];
const work = resolve('build/n29elec/filmw'); mkdirSync(work, { recursive: true });
const stage = resolve(work, 'stage'), source = resolve('project'), dst = resolve(stage, 'project');
mkdirSync(dst, { recursive: true });
for (const f of readdirSync(source, { withFileTypes: true })) {
  const a = resolve(source, f.name), b = resolve(dst, f.name);
  if (f.isDirectory()) { try { symlinkSync(a, b, 'junction'); } catch {} }
  else if (f.name !== 'project.godot') copyFileSync(a, b);
}
writeFileSync(resolve(dst, 'project.godot'), readFileSync(resolve(source, 'project.godot'), 'utf8')
  .replace(/viewport_width=\d+/, 'viewport_width=1920').replace(/viewport_height=\d+/, 'viewport_height=1080'));
const avi = resolve(work, 'take.avi').split(String.fromCharCode(92)).join('/');
const args = ['--rendering-driver','d3d12','--windowed','--position','0,1440','--resolution','1920x1080','--fixed-fps','60','--audio-driver','Dummy','--write-movie',avi,'-s','C:/Users/kazda/kiro/gtai-wt/n23-street/build/n29elec/film_entry.gd'];
const r = await runGodot(args, { cwd: stage, ensure: false, headless: false, timeoutMs: 900000, name: 'n29-elec-film' });
writeFileSync(resolve(work, 'film.log'), r.out);
console.log('godot exit', r.status, r.timedOut, r.out.split('\n').filter(l=>/film:|ERROR/.test(l)).join('\n'));
execFileSync('ffmpeg', ['-hide_banner','-loglevel','error','-y','-ss','0.5','-i',avi,'-c:v','libx264','-crf','18','-pix_fmt','yuv420p','-r','60',resolve(out)]);
console.log('wrote', out);
