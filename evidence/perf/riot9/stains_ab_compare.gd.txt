extends SceneTree
## Scratch (perf-city): CorpseStains printed with CircleFan.batched false vs true, exact compare.

var _vp: SubViewport
var _stains: CorpseStains
var _n: int = 0
var _imgs: Array[Image] = []


func _initialize() -> void:
	_vp = SubViewport.new()
	_vp.size = Vector2i(1280, 720)
	_vp.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(_vp)
	var cam: Node2D = Node2D.new()
	cam.position = Vector2(-4800.37, -2900.61) * 1.5 + Vector2(640, 360)
	cam.scale = Vector2.ONE * 1.5
	_vp.add_child(cam)
	var cb: CombatBlood = CombatBlood.new()
	_stains = CorpseStains.new(16, Color(0.64, 0.02, 0.035), Color.TRANSPARENT)
	_stains.material = cb.print_material()
	cam.add_child(_stains)
	var dead: Dictionary[int, Vector2] = {}
	for i: int in 16:
		var h: float = fposmod(sin(i * 91.3458 + 7.13) * 47453.5453, 1.0)
		var h2: float = fposmod(sin(i * 12.9898 + 1.7) * 43758.5453, 1.0)
		dead[i] = Vector2(4800.0 + (i % 4) * 100.0 + h * 40.0, 2900.0 + floorf(i / 4.0) * 100.0 + h2 * 40.0)
		_stains.on_bled_body(i, Vector2.from_angle(h * TAU))
	_stains.step_bodies(1, dead)
	_stains.step_bodies(16, dead)  # scales 0.6 for all: then vary per soak
	var k: int = 0
	for c: Node in _stains.get_children():
		(c as Node2D).scale = Vector2.ONE * (0.3 + 0.05 * k)
		k += 1
	CircleFan.batched = false


func _process(_delta: float) -> bool:
	_n += 1
	if _n == 6 or _n == 12:
		_imgs.append(_vp.get_texture().get_image())
		if _n == 6:
			CircleFan.batched = true
			for c: Node in _stains.get_children():
				(c as CanvasItem).queue_redraw()
	if _n < 12:
		return false
	var a: Image = _imgs[0]
	var b: Image = _imgs[1]
	var diff: int = 0
	var red: int = 0
	for y: int in a.get_height():
		for x: int in a.get_width():
			var pa: Color = a.get_pixel(x, y)
			if pa != b.get_pixel(x, y):
				diff += 1
			if pa.r > pa.g + 0.2:
				red += 1
	var _e1: Error = a.save_png(OS.get_environment("TEMP") + "/circprobe/stains_old.png")
	var _e2: Error = b.save_png(OS.get_environment("TEMP") + "/circprobe/stains_new.png")
	printerr("STAINSAB differing %d of %d, blood pixels %d" % [diff, a.get_width() * a.get_height(), red])
	quit(0)
	return true
